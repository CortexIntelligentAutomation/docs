{{/* Make sure that we are enclosed within a tabpane shortcode block */}} {{ with $.Parent }} {{- if ne $.Parent.Name "tabpane" -}} {{- errorf "tab.html: tab must be used within a tabpane block" -}} {{- end -}} {{- end -}} {{/* Prefill header if not given as parameter */}} {{ $header := default (printf "Tab %v" ( add $.Ordinal 1)) (.Get "header") }} {{/* store all tab info in dict tab */}} {{ $tab := dict "header" $header }} {{ with $.Get "lang" }} {{ $tab = merge $tab (dict "language" ($.Get "lang")) }} {{ end }} {{ with $.Get "highlight" }} {{ $tab = merge $tab (dict "highlight" ($.Get "highlight")) }} {{ end }} {{ with $.Inner }} {{/* Trim leading newlines and trailing whitespace from .Inner, this avoids spurious lines during syntax highlighting. \r is trimmed as well as \n, as a file with Windows line endings would otherwise leave a blank first line, and trailing spaces are trimmed as the indentation of the closing tab shortcode would otherwise leave a blank last line */}} {{ $tab = merge $tab (dict "content" (strings.TrimRight " \t\r\n" (strings.TrimLeft "\r\n" $.Inner))) }} {{ end }} {{/* add dict tab to parent's scratchpad */}} {{ with .Parent }} {{- $.Parent.Scratch.SetInMap "tabs" (printf "%v" $.Ordinal) $tab -}} {{ end }}
